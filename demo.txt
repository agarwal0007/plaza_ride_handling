# ===================================================================
# Ride Hailing CLI demo
# Run with: mvn -q compile exec:java < demo.txt
# ===================================================================

# --- Setup: two riders ---
add_user u1 Alice
add_user u2 Bob

# ===================================================================
# Scenario 1: happy path - a hatchback is booked and the ride ends
# with exactly the tiered fare worked out in the spec:
# 7km hatchback = 20 + 24 + 10 = Rs 54
# ===================================================================
add_driver d1 Carl HATCHBACK KA01AB1111 0 0
book_ride r1 u1 HATCHBACK 0 0 7 0
end_ride r1

# ===================================================================
# Scenario 2: minimum fare - a 1km hatchback ride's tiered fare
# (Rs 10) is below the Rs 50 minimum, so the floor wins
# ===================================================================
add_driver d2 Diana HATCHBACK KA02CD2222 100 100
book_ride r2 u2 HATCHBACK 100 100 101 100
end_ride r2

# ===================================================================
# Scenario 3: hatchback -> sedan free upgrade - no hatchback is
# within radius of this pickup, only a sedan is, so the sedan gets
# assigned but the rider is still billed at hatchback rates (same
# Rs 54 as scenario 1, for the same 7km distance)
# ===================================================================
add_driver d3 Evan SEDAN KA03EF3333 200 200
book_ride r3 u1 HATCHBACK 200 200 207 200
end_ride r3

# ===================================================================
# Scenario 4: no driver in radius - the only driver here is 50 units
# away, far outside the default 5-unit radius, so booking fails
# ===================================================================
add_driver d4 Frank SEDAN KA04GH4444 300 300
book_ride r4 u2 SEDAN 300 350 301 350

# ===================================================================
# Scenario 5: invalid coupon - a driver (d4, still available since
# scenario 4 failed) is right at this pickup, but booking with an
# unknown coupon code fails outright, releasing that driver again
# ===================================================================
book_ride r5 u2 SEDAN 300 300 301 300 NOPE

# ===================================================================
# Scenario 6: unknown coupon TYPE - add_coupon now parses
# <code> <TYPE> <args...> through DiscountStrategyFactory. FLAT isn't
# a registered type (only PERCENT is), so adding it fails and no
# coupon is created - but the CLI doesn't crash
# ===================================================================
add_coupon BADTYPE FLAT 25

# ===================================================================
# Scenario 7: coupon with a cap - a 50% coupon on a Rs 54 fare would
# be Rs 27, but this coupon is capped at Rs 10, so only Rs 10 is
# actually discounted
# ===================================================================
add_coupon CAPPED PERCENT 50 10
add_driver d5 Grace HATCHBACK KA05IJ5555 400 400
book_ride r6 u1 HATCHBACK 400 400 407 400 CAPPED
end_ride r6

# ===================================================================
# Scenario 8: free cancellation - cancelling right after booking is
# well inside the 2-minute default grace period, so there's no fee
# ===================================================================
add_user u3 Ivy
add_driver d6 Henry SEDAN KA06KL6666 500 500
book_ride r7 u3 SEDAN 500 500 505 500
cancel_ride r7

# ===================================================================
# Scenario 9: paid cancellation - this can't be demonstrated live in
# a one-shot script, because cancellation timing comes from the real
# system clock (java.time.Clock.systemDefaultZone() in Main), and
# this script runs in well under 2 minutes. To see the Rs 25 fee:
#   book_ride r8 u3 SEDAN 500 500 505 500   (after cancelling r7 above)
#   (wait more than 2 real minutes)
#   cancel_ride r8                           -> "Fee: Rs 25.00"
# RideServiceCancellationTest covers this deterministically instead,
# using a test-only MutableClock that can jump time forward instantly.
# ===================================================================

# ===================================================================
# Scenario 10: surge pricing - two pre-existing ongoing rides at this
# pickup make demand = 2; only one SEDAN driver is free there, so
# supply = 1 and ratio = demand / supply = 2 -> the 1.5x high band.
# Under the DEFAULT run (no -Dsurge flag) surge is off, so this books
# and ends with no surge adjustment either way. Re-run this same file
# with -Dsurge=demand to see "(Surge: 1.50x)" on the booking line and
# a non-zero surge adjustment on the fare breakdown:
#   mvn -q compile exec:java -Dsurge=demand < demo.txt
# ===================================================================
add_user u4 Jack
add_user u5 Karen
add_user u6 Leo
add_driver surgeBusy1 Mona SEDAN KA07MN7777 600 600
add_driver surgeBusy2 Nora SEDAN KA08NO8888 600 600
book_ride busyA u4 SEDAN 600 600 601 600
book_ride busyB u5 SEDAN 600 600 601 600
add_driver surgeFree Oscar SEDAN KA09OP9999 600 600
book_ride r8 u6 SEDAN 600 600 607 600
end_ride r8

# ===================================================================
# Scenario 11: rated matching - -Dmatching is a startup flag too, so
# it can't be toggled mid-script either. To see it change which
# driver gets picked, run this separately:
#   printf 'add_user ur Pat\nadd_driver close C SEDAN KA10 1 0 3.0\nadd_driver far F SEDAN KA11 4 0 4.9\nbook_ride rr ur SEDAN 0 0 1 1\nexit\n' \
#     | mvn -q compile exec:java
#   (books "close" - nearer, lower rated, the default -Dmatching=nearest)
#   printf 'add_user ur Pat\nadd_driver close C SEDAN KA10 1 0 3.0\nadd_driver far F SEDAN KA11 4 0 4.9\nbook_ride rr ur SEDAN 0 0 1 1\nexit\n' \
#     | mvn -q compile exec:java -Dmatching=rated
#   (books "far" instead - farther away, but its 4.9 rating beats close's 3.0)
# ===================================================================

# ===================================================================
# Scenario 12: histories - now covering a completed ride, an ongoing
# one, and a cancelled one, for both a user and a driver
# ===================================================================
user_history u1
user_history u3
driver_history d1
driver_history d6

help
exit
